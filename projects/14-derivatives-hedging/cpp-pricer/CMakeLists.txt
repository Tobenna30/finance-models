cmake_minimum_required(VERSION 3.16)
project(refinery_pricer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pricer pricer.cpp)
target_compile_options(pricer PRIVATE -Wall -Wextra -Wpedantic)
add_executable(refinery_pricer main.cpp)
target_link_libraries(refinery_pricer PRIVATE pricer)
add_executable(native_test native_test.cpp)
target_link_libraries(native_test PRIVATE pricer)
enable_testing()
add_test(NAME pricing_invariants COMMAND native_test)
